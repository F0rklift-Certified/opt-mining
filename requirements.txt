# Supported Python: 3.13 only (>=3.13,<3.14)
#
# The project standardises on Python 3.13 — this is the single interpreter that
# CI (.github/workflows/ci.yml) tests, so support matches verification.
#
# DO NOT run this pipeline on Python 3.14. The grid builder
# (pipeline/grid/generate.py, S1-02) produces degenerate zero-area cells on
# CPython 3.14.x — every analysis cell collapses to a single point, which
# breaks the whole grid and every feature layer that joins to it. The exact
# same source, numpy (2.2.6) and shapely (2.1.2) work correctly on Python
# 3.13; the fault is in the 3.14 interpreter, not this code. See pyproject.toml
# (requires-python) which enforces this for pip installs.
pandas==2.2.2
requests==2.32.3

# Task 1 — Wind Resource (Global Wind Atlas GeoTIFF handling)
# Bumped from 1.4.3 to 1.4.4 (same 1.4.x line): Python 3.14 has no 1.4.3 wheel
# and there is no local gdal-config to build from source. 1.4.4 ships a 3.14 wheel.
rasterio==1.4.4
numpy==2.2.6

# Sprint 1 — Grid generation and feature layers (S1-02 onwards)
# Introduced at S1-02 (Finalise Common Analysis Cell) because generating ~47k

# will use these for spatial joins, overlay, and GeoPackage I/O.
geopandas==1.1.4
shapely==2.1.2
pyogrio==0.13.0
pyproj==3.7.2
